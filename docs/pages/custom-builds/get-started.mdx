import React, { useState } from "react";
import {
  View,
  Text,
  StyleSheet,
  FlatList,
  Image,
  TouchableOpacity,
  TextInput,
  SafeAreaView,
  ScrollView,
} from "react-native";

const starterPosts = [
  {
    id: "1",
    user: "You",
    caption: "Morning coffee together ☕",
    image:
      "https://images.unsplash.com/photo-1495474472287-4d71bcdd2085?q=80&w=1200&auto=format&fit=crop",
    time: "2h ago",
  },
  {
    id: "2",
    user: "Girlfriend",
    caption: "Movie night 🍿",
    image:
      "https://images.unsplash.com/photo-1517604931442-7e0c8ed2963c?q=80&w=1200&auto=format&fit=crop",
    time: "5h ago",
  },
];

export default function App() {
  const [posts, setPosts] = useState(starterPosts);
  const [caption, setCaption] = useState("");

  const addPost = () => {
    if (!caption) return;

    const newPost = {
      id: Date.now().toString(),
      user: "You",
      caption,
      image:
        "https://images.unsplash.com/photo-1524504388940-b1c1722653e1?q=80&w=1200&auto=format&fit=crop",
      time: "Just now",
    };

    setPosts([newPost, ...posts]);
    setCaption("");
  };

  return (
    <SafeAreaView style={styles.container}>
      <ScrollView showsVerticalScrollIndicator={false}>
        <View style={styles.header}>
          <Text style={styles.title}>Daily Couple Moments 💖</Text>
          <Text style={styles.subtitle}>
            Share everyday memories together.
          </Text>
        </View>

        <View style={styles.uploadCard}>
          <Text style={styles.cardTitle}>Post Today's Moment</Text>

          <TextInput
            placeholder="What are you doing today?"
            placeholderTextColor="#999"
            style={styles.input}
            value={caption}
            onChangeText={setCaption}
          />

          <TouchableOpacity style={styles.uploadButton}>
            <Text style={styles.uploadButtonText}>📷 Choose Picture</Text>
          </TouchableOpacity>

          <TouchableOpacity style={styles.postButton} onPress={addPost}>
            <Text style={styles.postButtonText}>Share Moment ✨</Text>
          </TouchableOpacity>
        </View>

        <Text style={styles.feedTitle}>Your Shared Feed</Text>

        <FlatList
          data={posts}
          keyExtractor={(item) => item.id}
          scrollEnabled={false}
          renderItem={({ item }) => (
            <View style={styles.postCard}>
              <Image source={{ uri: item.image }} style={styles.postImage} />

              <View style={styles.postContent}>
                <View style={styles.postHeader}>
                  <View>
                    <Text style={styles.postUser}>{item.user}</Text>
                    <Text style={styles.postTime}>{item.time}</Text>
                  </View>

                  <Text style={styles.heart}>❤️</Text>
                </View>

                <Text style={styles.caption}>{item.caption}</Text>
              </View>
            </View>
          )}
        />
      </ScrollView>
    </SafeAreaView>
  );
}
